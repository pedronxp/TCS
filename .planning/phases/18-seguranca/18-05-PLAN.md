## Tasks

### Task 1 — utils/auditUtils.ts (novo arquivo)

```ts
import { supabase } from './supabase';

type AuditAcao =
  | 'laudo_gerado'
  | 'vistoria_acessada'
  | 'token_criado'
  | 'token_usado'
  | 'login_falhou'
  | 'usuario_criado'
  | 'vistoria_criada'
  | 'agendamento_criado';

export async function logAudit(
  acao: AuditAcao,
  atorUid: string | null,
  atorRole: string | null,
  detalhes?: Record<string, any>
): Promise<void> {
  try {
    await supabase.from('audit_logs').insert({
      acao,
      ator_uid: atorUid,
      ator_role: atorRole,
      detalhes: detalhes ? JSON.stringify(detalhes) : null,
      criado_em: new Date().toISOString(),
    });
  } catch {
    // Nunca crashar por falha de audit log
  }
}
```
