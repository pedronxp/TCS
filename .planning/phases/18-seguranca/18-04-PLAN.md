## Tasks

### Task 1 — utils/validationUtils.ts (novo arquivo)

```ts
export const MAX_LENGTHS = {
  nome: 100,
  endereco: 200,
  municipio: 80,
  cep: 9,
  observacoes: 2000,
  token: 64,
  email: 254,
};

export function sanitizarTexto(texto: string): string {
  return texto
    .trim()
    .replace(/[\x00-\x1F\x7F]/g, '') // remover control chars
    .substring(0, 2000);
}

export function validarEmail(email: string): boolean {
  return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email) && email.length <= MAX_LENGTHS.email;
}

export function validarCep(cep: string): boolean {
  return /^\d{5}-?\d{3}$/.test(cep.replace(/\D/g, '').replace(/(\d{5})(\d{3})/, '$1-$2'));
}

export function validarNome(nome: string): { valido: boolean; erro?: string } {
  const limpo = nome.trim();
  if (limpo.length === 0) return { valido: false, erro: 'Nome obrigatório' };
  if (limpo.length > MAX_LENGTHS.nome) return { valido: false, erro: `Máximo ${MAX_LENGTHS.nome} caracteres` };
  if (/<[^>]*>/.test(limpo)) return { valido: false, erro: 'Caracteres inválidos' };
  return { valido: true };
}

export function validarEndereco(endereco: string): { valido: boolean; erro?: string } {
  const limpo = endereco.trim();
  if (limpo.length === 0) return { valido: false, erro: 'Endereço obrigatório' };
  if (limpo.length > MAX_LENGTHS.endereco) return { valido: false, erro: `Máximo ${MAX_LENGTHS.endereco} caracteres` };
  if (/<[^>]*>/.test(limpo)) return { valido: false, erro: 'Caracteres inválidos' };
  return { valido: true };
}
```
