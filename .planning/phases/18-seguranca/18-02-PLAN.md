## Goal

Criar tabela `rate_limits` no Supabase e implementar verificação no app antes de ações de alto custo: geração de PDF, criação de vistoria e tentativas de login.

## Approach

- Tabela `rate_limits` com `uid`, `action`, `window_start`, `count`
- Verificação via `supabase.rpc('check_rate_limit')` antes de cada ação
- Função PL/pgSQL atômica com upsert + verificação

## Tasks

### Task 1 — Tabela e função rate_limits no Supabase

```sql
CREATE TABLE IF NOT EXISTS rate_limits (
  uid UUID NOT NULL,
  action TEXT NOT NULL,
  window_start TIMESTAMPTZ NOT NULL,
  count INTEGER DEFAULT 1,
  PRIMARY KEY (uid, action, window_start)
);

ALTER TABLE rate_limits ENABLE ROW LEVEL SECURITY;

-- Somente o próprio usuário e service_role acessam
CREATE POLICY "usuario_ve_proprios_limites" ON rate_limits
  FOR ALL USING (uid = auth.uid() OR auth.role() = 'service_role');

-- Função que verifica e incrementa atomicamente
CREATE OR REPLACE FUNCTION check_rate_limit(
  p_uid UUID,
  p_action TEXT,
  p_max_count INTEGER,
  p_window_seconds INTEGER
)
RETURNS BOOLEAN AS $$
DECLARE
  v_window TIMESTAMPTZ;
  v_count INTEGER;
BEGIN
  -- Janela atual (truncada ao início da janela)
  v_window := date_trunc('second', NOW()) - (
    EXTRACT(EPOCH FROM NOW())::INTEGER % p_window_seconds
  ) * INTERVAL '1 second';

  -- Upsert atômico
  INSERT INTO rate_limits (uid, action, window_start, count)
    VALUES (p_uid, p_action, v_window, 1)
    ON CONFLICT (uid, action, window_start)
    DO UPDATE SET count = rate_limits.count + 1
    RETURNING count INTO v_count;

  -- Limpar janelas antigas (manutenção inline)
  DELETE FROM rate_limits
    WHERE uid = p_uid AND action = p_action
      AND window_start < NOW() - (p_window_seconds * 2) * INTERVAL '1 second';

  RETURN v_count <= p_max_count;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;
```

### Task 2 — utils/rateLimitUtils.ts (novo arquivo)

```ts
import { supabase } from './supabase';

type RateLimitAction = 'gerar_pdf' | 'criar_vistoria' | 'login';

const LIMITS: Record<RateLimitAction, { max: number; windowSeconds: number; msg: string }> = {
  gerar_pdf:      { max: 10, windowSeconds: 3600,  msg: 'Limite de 10 PDFs por hora atingido. Tente novamente em instantes.' },
  criar_vistoria: { max: 30, windowSeconds: 86400, msg: 'Limite de 30 vistorias por dia atingido.' },
  login:          { max: 5,  windowSeconds: 900,   msg: 'Muitas tentativas de login. Aguarde 15 minutos.' },
};

export async function checkRateLimit(uid: string, action: RateLimitAction): Promise<{ allowed: boolean; message?: string }> {
  try {
    const limit = LIMITS[action];
    const { data, error } = await supabase.rpc('check_rate_limit', {
      p_uid: uid,
      p_action: action,
      p_max_count: limit.max,
      p_window_seconds: limit.windowSeconds,
    });

    if (error) return { allowed: true }; // fail-open: não bloquear se verificação falhar
    return data ? { allowed: true } : { allowed: false, message: limit.msg };
  } catch {
    return { allowed: true }; // fail-open
  }
}
```

### Task 5 — app/(auth)/index.tsx: verificar rate limit no login

Rate limit de login baseado em email (não uid, pois usuário não está autenticado):
- Simplificado: usar IP via `Linking` — no app mobile não há IP real disponível facilmente
- Implementar via AsyncStorage: contar tentativas locais nos últimos 15 min
- Se count >= 5: desabilitar botão e mostrar countdown

```ts
// utils/loginRateLimit.ts
import AsyncStorage from '@react-native-async-storage/async-storage';

const KEY = (email: string) => `login_attempts_${email.toLowerCase()}`;

export async function recordLoginAttempt(email: string): Promise<boolean> {
  const raw = await AsyncStorage.getItem(KEY(email));
  const data = raw ? JSON.parse(raw) : { count: 0, since: Date.now() };

  // Reset janela se passou 15 min
  if (Date.now() - data.since > 15 * 60 * 1000) {
    data.count = 0;
    data.since = Date.now();
  }

  data.count += 1;
  await AsyncStorage.setItem(KEY(email), JSON.stringify(data));
  return data.count <= 5;
}

export async function getLoginBlockedUntil(email: string): Promise<number | null> {
  const raw = await AsyncStorage.getItem(KEY(email));
  if (!raw) return null;
  const data = JSON.parse(raw);
  if (data.count < 5) return null;
  const unblockAt = data.since + 15 * 60 * 1000;
  if (Date.now() > unblockAt) return null;
  return unblockAt;
}
```
