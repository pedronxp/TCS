## Goal

Melhorar a mensagem de compartilhamento (WhatsApp/email) para incluir: protocolo sequencial, endereço completo, nível de risco, nome do agente, data/hora, e "Secretaria de Origem: CidadeX" somente quando o agente é de município diferente da vistoria.

## Tasks

### Task 1 — utils/shareUtils.ts (novo arquivo)

```ts
import { riscoLabel } from './riscoUtils';

export interface ShareMessageParams {
  protocolo?: string | null;
  endereco: string;
  municipio: string;
  municipio_agente?: string | null;
  nivelRisco: string;
  agenteNome: string;
  dataVistoria: string;
}

export function buildShareMessage(p: ShareMessageParams): string {
  const risco = riscoLabel(p.nivelRisco);
  const data = new Date(p.dataVistoria).toLocaleDateString('pt-BR', {
    day: '2-digit', month: '2-digit', year: 'numeric',
    hour: '2-digit', minute: '2-digit',
  });

  const crossMunicipio =
    p.municipio_agente &&
    p.municipio_agente.trim().toLowerCase() !== p.municipio.trim().toLowerCase();

  const linhas = [
    `📋 *TCS — Relatório de Risco*`,
    p.protocolo ? `🔖 Protocolo: ${p.protocolo}` : null,
    ``,
    `📍 *${p.endereco}*`,
    `🏙️ Município: ${p.municipio}`,
    crossMunicipio ? `🏛️ Secretaria de Origem: ${p.municipio_agente}` : null,
    ``,
    `⚠️ Nível de Risco: *${risco}*`,
    ``,
    `👤 Agente: ${p.agenteNome}`,
    `📅 Data: ${data}`,
    ``,
    `_Documento gerado pelo TCS — Relatório de Risco_`,
  ].filter(l => l !== null).join('\n');

  return linhas;
}
```
