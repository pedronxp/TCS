### Task 3 — Criar função + trigger para gerar protocolo sequencial

```sql
CREATE OR REPLACE FUNCTION gerar_protocolo_seq()
RETURNS TRIGGER AS $$
DECLARE
  v_codigo TEXT;
  v_ano INTEGER;
  v_seq INTEGER;
BEGIN
  -- Buscar código do município (fallback: primeiras 3 letras em maiúsculo)
  SELECT COALESCE(codigo, UPPER(LEFT(NEW.municipio, 3)))
    INTO v_codigo
    FROM municipios
    WHERE nome ILIKE NEW.municipio
    LIMIT 1;

  IF v_codigo IS NULL THEN
    v_codigo := UPPER(LEFT(NEW.municipio, 3));
  END IF;

  v_ano := EXTRACT(YEAR FROM NOW())::INTEGER;

  -- Incrementar contador atomicamente
  INSERT INTO contadores_protocolo (municipio_codigo, ano, ultimo_seq)
    VALUES (v_codigo, v_ano, 1)
    ON CONFLICT (municipio_codigo, ano)
    DO UPDATE SET ultimo_seq = contadores_protocolo.ultimo_seq + 1
    RETURNING ultimo_seq INTO v_seq;

  NEW.protocolo := 'TCS-' || v_codigo || '-' || v_ano || '-' || LPAD(v_seq::TEXT, 5, '0');
  NEW.protocolo_seq := v_seq;

  RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

CREATE OR REPLACE TRIGGER trg_gerar_protocolo
  BEFORE INSERT ON vistorias
  FOR EACH ROW
  WHEN (NEW.protocolo IS NULL)
  EXECUTE FUNCTION gerar_protocolo_seq();
```
