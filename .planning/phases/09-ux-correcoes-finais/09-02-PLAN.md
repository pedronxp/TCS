export interface LogEntry {
  id: number;          // PK SQLite
  level: LogLevel;     // 'info' | 'warn' | 'error'  (NÃO 'aviso' nem 'nivel')
  category: LogCategory; // módulo de origem
  message: string;     // texto do log  (NÃO 'descricao' nem 'mensagem')
  data: string | null; // JSON serializado ou null
  criado_em: string;   // ISO 8601  (NÃO 'criadoEm')
}

export function getLogs(options?: {
  limit?: number;

export function countLogsByLevel(): { info: number; warn: number; error: number };
